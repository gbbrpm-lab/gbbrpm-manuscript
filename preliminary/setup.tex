\makeatletter

% ---------- Page styles and pagination (Article 8) ----------
% Preliminary pages: lowercase roman, bottom center, 1.25 cm from the bottom edge.
\fancypagestyle{prelim}{%
  \fancyhf{}%
  \fancyfoot[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}}
\setlength{\footskip}{1.25cm}% bottom margin 2.5 cm minus 1.25 cm


% Continuation pages of the contents and lists repeat the column headings (3.5.12).
\newcommand{\prelim@defcont}[2]{%
  \fancypagestyle{#1}{%
    \fancyhf{}%
    \fancyfoot[C]{\thepage}%
    \renewcommand{\headrulewidth}{0pt}%
    \fancyhead[R]{\hb@xt@\@pnumwidth{\hss\prelimcolhead{Page}\hss}}%
    \if\relax\detokenize{#2}\relax\else
      \fancyhead[L]{\hb@xt@\prelimnumwidth{\hss\prelimcolhead{#2}\hss}}%
    \fi}}
\newlength{\prelimnumwidth}\setlength{\prelimnumwidth}{2cm}
\prelim@defcont{prelimcont}{}
\prelim@defcont{prelimcontT}{Table}
\prelim@defcont{prelimcontF}{Figure}
\prelim@defcont{prelimcontE}{Equation}
\newcommand{\prelimnext}{\clearpage\pagestyle{prelim}}

% Main body: page number in the upper right, 2.5 cm from the right edge and
% 1.25 cm from the top (Section 8.2.b). Defined as its own style so it does not
% inherit the bottom-center footer of the preliminary pages.
\fancypagestyle{bodystyle}{%
  \fancyhf{}%
  \fancyhead[R]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}}

% ---------- Page title and column heading helpers ----------
\newcommand{\prelimheading}[1]{{\centering\bfseries\MakeUppercase{#1}\par}}
\newcommand{\prelimcolhead}[1]{\underline{\MakeUppercase{#1}}}

% ---------- Table of contents, lists (Sections 3.5 and 3.6) ----------
% No dot leaders; page numbers centered under the PAGE heading (3.5.5, 3.5.6).
% Double spacing between entries (one blank line), single spacing within
% an entry (3.5.8, 3.6.10).
\renewcommand{\@pnumwidth}{1cm}
\renewcommand{\@tocrmarg}{1.25cm}
\renewcommand*\@dottedtocline[5]{%
  \ifnum #1>\c@tocdepth \else
    \vskip\baselineskip % one blank line between entries (3.5.8)
    {\leftskip #2\relax \rightskip \@tocrmarg \parfillskip -\rightskip
     \parindent #2\relax\@afterindenttrue
     \interlinepenalty\@M
     \leavevmode
     \@tempdima #3\relax
     \advance\leftskip \@tempdima \null\nobreak\hskip -\leftskip
     {#4}\nobreak\hfill\nobreak
     \hb@xt@\@pnumwidth{\hss #5\hss}\par}%
  \fi}

% Main headings: uppercase, no number, flush left; same gap as other entries.
\renewcommand*\l@section[2]{%
  \begingroup
    \let\numberline\@gobble
    \let\hyper@linkstart\@gobbletwo
    \let\hyper@linkend\@empty
    \@dottedtocline{1}{0cm}{0cm}{\MakeUppercase{#1}}{#2}%
  \endgroup}
% Each lower level is indented 1.25 cm from the previous one (3.5.10).
\renewcommand*\l@subsection{\@dottedtocline{2}{1.25cm}{1.25cm}}
\renewcommand*\l@subsubsection{\@dottedtocline{3}{2.5cm}{1.5cm}}
\setcounter{tocdepth}{3}

% Lists of tables, figures, equations: number column centered under its heading.
% One blank line between entries (3.6.10).
\newcommand*\prelim@listline[2]{%
  \vskip\baselineskip
  {\leftskip \dimexpr\prelimnumwidth+0.2cm\relax \rightskip \@tocrmarg
   \parfillskip -\rightskip \parindent 0pt \interlinepenalty\@M
   \leavevmode
   \def\numberline##1{\hb@xt@\prelimnumwidth{\hss##1\hss}\hskip 0.2cm}%
   \hskip -\leftskip #1\nobreak\hfill\nobreak
   \hb@xt@\@pnumwidth{\hss #2\hss}\par}}
\renewcommand*\l@figure{\prelim@listline}
\renewcommand*\l@table{\prelim@listline}
\newcommand*\l@equation{\prelim@listline}

% Equation tag with an automatic entry for the List of Equations.
% Usage: \eqtag{3-1}  or  \eqtag[Short description]{3-1}
\input{preliminary/equation_descriptions}
\renewcommand{\eqtag}[2][]{%
  \tag{\hypertarget{eqtag.#2}{\textit{Equation #2}}}%
  \begingroup
  \edef\@currentHref{eqtag.#2}%
  \def\prelim@eqdesc{#1}%
  \ifx\prelim@eqdesc\@empty
    \ifcsname eqdesc@#2\endcsname
      \edef\prelim@eqdesc{\csname eqdesc@#2\endcsname}%
    \else
      \def\prelim@eqdesc{Equation #2}%
    \fi
  \fi
  \addcontentsline{loe}{equation}{\protect\numberline{#2}\prelim@eqdesc}%
  \endgroup}


% Heading row: column titles over the number and page columns.
\newcommand{\prelim@colrow}[1]{%
  \noindent\hb@xt@\textwidth{%
  \if\relax\detokenize{#1}\relax\else
    \hb@xt@\prelimnumwidth{\hss\prelimcolhead{#1}\hss}\fi
  \hfill\hb@xt@\@pnumwidth{\hss\prelimcolhead{Page}\hss}}\par}

% ---------- Table of Contents page (Section 3.5) ----------
% Usage in the main file:  \prelimtableofcontents
% Title at the top; "PAGE" 4 spaces below it (3 blank lines, 3.5.3);
% first entry 3 spaces below "PAGE" (2 blank lines, 3.5.4: the explicit
% skip below plus the entry's own gap).
\newcommand{\prelimtableofcontents}{%
  \prelimnext
  \thispagestyle{prelim}\pagestyle{prelimcont}% PAGE heading repeats (3.5.12)
  \prelimheading{Table of Contents}%
  \vskip3\baselineskip
  \prelim@colrow{}%
  \vskip\baselineskip
  \@starttoc{toc}%
  \prelimnext}

% ---------- Appendices in the table of contents ----------
% APPENDICES is a main heading; each appendix sits one level below it and
% its lettered subsections one level further (switched on by \apptocmode).
\providecommand*{\toclevel@appendixtitle}{2}
\newcommand*\l@appendixtitle[2]{{\raggedright\@dottedtocline{2}{1.25cm}{0cm}{#1}{#2}}}
\newcommand*\apptocmode{\renewcommand*\l@subsection{\@dottedtocline{3}{2.5cm}{1.5cm}}}

% ---------- Entries for REFERENCES in the table of contents ----------
\apptocmd{\thebibliography}{\addcontentsline{toc}{section}{References}}{}{}

\makeatother
